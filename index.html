{%- endcapture -%} {% include section.html id="news" title="News" content=news_board %} {%- endif %} {%- comment -%} Projects are split into pages of `projects_per_page` here, so the grouping exists in the HTML rather than being assembled by script. Every project and every modal is in the page, so nothing is hidden from search engines. js/project-pager.js slides the track sideways between pages. Without it the pages simply stack and all projects are visible, which is why the sliding CSS is gated behind the .is-paged class the script adds. Modal ids are global, so the card index has to be the offset plus the position within the page, not the inner loop's own index. {%- endcomment -%} {%- assign per_page = site.projects_per_page | default: 6 -%} {%- assign total_projects = site.data.projects | size -%} {%- assign page_count = total_projects | divided_by: per_page -%} {%- assign leftover = total_projects | modulo: per_page -%} {%- if leftover > 0 -%}{%- assign page_count = page_count | plus: 1 -%}{%- endif -%} {%- capture projects_grid -%}
{%- for p in (1..page_count) -%} {%- assign offset = forloop.index0 | times: per_page -%}
{%- for project in site.data.projects offset: offset limit: per_page -%} {%- assign card_index = offset | plus: forloop.index0 -%} {% include project-card.html project=project index=card_index %} {%- endfor -%}
{%- endfor -%}
